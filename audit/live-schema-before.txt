
--- achievements ---
CREATE TABLE `achievements` (
  `achievement_id` int(10) unsigned NOT NULL AUTO_INCREMENT,
  `created_by_user_id` int(10) unsigned DEFAULT NULL,
  `title` varchar(200) NOT NULL,
  `description` text NOT NULL,
  `awardee` varchar(200) DEFAULT NULL COMMENT 'Person or group that received it',
  `category` varchar(100) DEFAULT NULL,
  `achievement_date` date DEFAULT NULL,
  `image` varchar(255) DEFAULT NULL COMMENT 'Path in storage/app/public/achievements',
  `approval_status` varchar(20) NOT NULL DEFAULT 'pending' COMMENT 'pending, approved, or rejected',
  `content_status` varchar(20) NOT NULL DEFAULT 'active' COMMENT 'active or archived',
  `rejection_reason` varchar(255) DEFAULT NULL,
  `reviewed_by_user_id` int(10) unsigned DEFAULT NULL,
  `reviewed_at` timestamp NULL DEFAULT NULL,
  `created_at` timestamp NULL DEFAULT current_timestamp(),
  `updated_at` timestamp NULL DEFAULT NULL,
  PRIMARY KEY (`achievement_id`),
  KEY `idx_achievements_created_by` (`created_by_user_id`),
  KEY `idx_achievements_reviewed_by` (`reviewed_by_user_id`),
  KEY `idx_achievements_status` (`approval_status`,`content_status`),
  KEY `idx_achievements_date` (`achievement_date`),
  CONSTRAINT `fk_achievements_created_by` FOREIGN KEY (`created_by_user_id`) REFERENCES `users` (`user_id`) ON DELETE SET NULL ON UPDATE CASCADE,
  CONSTRAINT `fk_achievements_reviewed_by` FOREIGN KEY (`reviewed_by_user_id`) REFERENCES `users` (`user_id`) ON DELETE SET NULL ON UPDATE CASCADE,
  CONSTRAINT `chk_achievements_approval` CHECK (`approval_status` in ('pending','approved','rejected')),
  CONSTRAINT `chk_achievements_content` CHECK (`content_status` in ('active','archived'))
) ENGINE=InnoDB AUTO_INCREMENT=2 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci

--- notifications ---
CREATE TABLE `notifications` (
  `id` char(36) NOT NULL COMMENT 'UUID generated by Laravel',
  `type` varchar(255) NOT NULL COMMENT 'Notification class name',
  `notifiable_type` varchar(255) NOT NULL COMMENT 'Always App\\Models\\User here',
  `notifiable_id` bigint(20) unsigned NOT NULL COMMENT 'The recipient users.user_id',
  `data` text NOT NULL COMMENT 'JSON: post type, post id, title, message, url',
  `read_at` timestamp NULL DEFAULT NULL COMMENT 'NULL means unread',
  `created_at` timestamp NULL DEFAULT NULL,
  `updated_at` timestamp NULL DEFAULT NULL,
  PRIMARY KEY (`id`),
  KEY `idx_notifiable` (`notifiable_type`,`notifiable_id`),
  KEY `idx_unread` (`notifiable_id`,`read_at`,`created_at`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci

--- officer_terms ---
CREATE TABLE `officer_terms` (
  `term_id` int(10) unsigned NOT NULL AUTO_INCREMENT,
  `user_id` int(10) unsigned DEFAULT NULL,
  `academic_year` varchar(9) NOT NULL COMMENT 'Format YYYY-YYYY, e.g. 2026-2027',
  `name` varchar(100) NOT NULL COMMENT 'Copied from the user so history stays accurate',
  `position` varchar(100) NOT NULL COMMENT 'Copied so the title is kept for that year',
  `photo` varchar(255) DEFAULT NULL COMMENT 'Path in storage/app/public/officer-photos, or an external URL',
  `sort_order` tinyint(3) unsigned NOT NULL DEFAULT 0,
  `created_at` timestamp NULL DEFAULT current_timestamp(),
  `updated_at` timestamp NULL DEFAULT NULL,
  PRIMARY KEY (`term_id`),
  UNIQUE KEY `uq_terms_user_year` (`user_id`,`academic_year`),
  KEY `idx_terms_year` (`academic_year`),
  KEY `idx_terms_user` (`user_id`),
  CONSTRAINT `fk_terms_user` FOREIGN KEY (`user_id`) REFERENCES `users` (`user_id`) ON DELETE SET NULL ON UPDATE CASCADE,
  CONSTRAINT `chk_terms_year_format` CHECK (`academic_year` regexp '^[0-9]{4}-[0-9]{4}$')
) ENGINE=InnoDB AUTO_INCREMENT=17 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci

--- users ---
CREATE TABLE `users` (
  `user_id` int(10) unsigned NOT NULL AUTO_INCREMENT,
  `student_number` varchar(20) NOT NULL,
  `name` varchar(100) NOT NULL,
  `year_level` varchar(30) NOT NULL,
  `program` varchar(100) NOT NULL,
  `institute` varchar(150) NOT NULL,
  `gender` varchar(20) NOT NULL,
  `contact_number` varchar(20) NOT NULL,
  `email` varchar(100) NOT NULL,
  `microsoft_id` varchar(255) DEFAULT NULL,
  `address` varchar(255) NOT NULL,
  `password` varchar(255) NOT NULL,
  `remember_token` varchar(100) DEFAULT NULL,
  `profile_picture` varchar(255) NOT NULL,
  `role` varchar(20) NOT NULL DEFAULT 'student',
  `officer_position` varchar(100) DEFAULT NULL,
  `updated_at` timestamp NULL DEFAULT current_timestamp() ON UPDATE current_timestamp(),
  `profile_completed_at` timestamp NULL DEFAULT NULL,
  PRIMARY KEY (`user_id`),
  UNIQUE KEY `email` (`email`),
  UNIQUE KEY `microsoft_id` (`microsoft_id`)
) ENGINE=InnoDB AUTO_INCREMENT=8 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci
